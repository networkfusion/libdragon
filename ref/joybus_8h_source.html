344inline void joybus_exec_cmd(
345 int port,
346 size_t send_len,
347 size_t recv_len,
348 const void *send_data,
349 void *recv_data
350)
351{
352 // Validate the desired Joybus port offset
353 assert((port >= 0) && (port < JOYBUS_PORT_COUNT));
354 // Ensure the send_len and recv_len fit in the Joybus operation block
355 assert((port + send_len + recv_len) < (JOYBUS_BLOCK_SIZE - 4));
356 // Allocate the Joybus operation block input and output buffers
357 uint8_t input[JOYBUS_BLOCK_SIZE] = {0};
358 uint8_t output[JOYBUS_BLOCK_SIZE] = {0};
359 // Skip commands on ports before the desired port offset
360 size_t i = port;
361 // Set the command metadata
362 input[i++] = send_len;
363 input[i++] = recv_len;
364 // Copy the send_data into the input buffer
365 memcpy(&input[i], send_data, send_len);
366 i += send_len + recv_len;
367 // Close out the Joybus operation block
368 input[i] = 0xFE;
369 input[sizeof(input) - 1] = 0x01;
370 // Execute the Joybus operation synchronously
371 joybus_exec(input, output);
372 // Copy recv_data from the output buffer
373 memcpy(recv_data, &output[i - recv_len], recv_len);
374}
