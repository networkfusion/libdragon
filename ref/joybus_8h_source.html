304inline void joybus_exec_cmd(
305 int port,
306 size_t send_len,
307 size_t recv_len,
308 const void *send_data,
309 void *recv_data
310)
311{
312 // Validate the desired Joybus port offset
313 assert((port >= 0) && (port < JOYBUS_PORT_COUNT));
314 // Ensure the send_len and recv_len fit in the Joybus operation block
315 assert((port + send_len + recv_len) < (JOYBUS_BLOCK_SIZE - 4));
316 // Allocate the Joybus operation block input and output buffers
317 uint8_t input[JOYBUS_BLOCK_SIZE] = {0};
318 uint8_t output[JOYBUS_BLOCK_SIZE] = {0};
319 // Skip commands on ports before the desired port offset
320 size_t i = port;
321 // Set the command metadata
322 input[i++] = send_len;
323 input[i++] = recv_len;
324 // Copy the send_data into the input buffer
325 memcpy(&input[i], send_data, send_len);
326 i += send_len + recv_len;
327 // Close out the Joybus operation block
328 input[i] = 0xFE;
329 input[sizeof(input) - 1] = 0x01;
330 // Execute the Joybus operation synchronously
331 joybus_exec(input, output);
332 // Copy recv_data from the output buffer
333 memcpy(recv_data, &output[i - recv_len], recv_len);
334}
