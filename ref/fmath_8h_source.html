106static inline float fm_truncf(float x) {
107 /* Notice that trunc.w.s is also emitted by the compiler when casting a
108 * float to int, but in this case we want a floating point result anyway,
109 * so it's useless to go back and forth a GPR. */
110 float yint, y;
111 __asm ("trunc.w.s %0,%1" : "=f"(yint) : "f"(x));
112 __asm ("cvt.s.w %0,%1" : "=f"(y) : "f"(yint));
113 return y;
114}

133static inline float fm_floorf(float x) {
134 float y = fm_truncf(x);
135 // After truncation, correct the negative numbers

253 // The following macros contain a special-case: when called with constant
254 // arguments, they fall back to the standard math.h version. This allows
255 // the compiler to compute an accurate result at compile time.
256 // For instance, if we find `sinf(sqrtf(2.0f))` in the source code,
257 // we expect that to be resolved at compile-time with the maximum accuracy,
258 // so it's important to generate code that calls the standard C function
259 // which is then intrinsified by the compiler.
260 #define fmodf(x, y) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? fmodf(x,y) : fm_fmodf(x,y))
261 #define sinf(x) (__builtin_constant_p(x) ? sinf(x) : fm_sinf(x))
262 #define cosf(x) (__builtin_constant_p(x) ? cosf(x) : fm_cosf(x))
263 #define sincosf(x,s,c) (__builtin_constant_p(x) ? sincosf(x,s,c) : fm_sincosf(x,s,c))
264 #define atan2f(y, x) ((__builtin_constant_p(x) && __builtin_constant_p(y)) ? atan2f(y, x) : fm_atan2f(y, x))
265#endif
