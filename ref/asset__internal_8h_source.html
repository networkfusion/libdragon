27__attribute__((used))
28static inline int asset_buf_size(int size, int cmp_size, int margin, int *cmp_offset_dst)
29{
30 // add 8 because the assembly decompressors do writes up to 8 bytes out-of-bounds,
31 // that could overwrite the input data.
32 margin += 8;
33 int bufsize = size + margin;
34 int cmp_offset = bufsize - cmp_size;
35 // Align the source buffer to 4 bytes, so that we can use 32-bit loads (required by shrinkler).
36 // Notice that we need at least 2-byte alignment anyway, for DMA.
37 while (cmp_offset & 3) {
38 cmp_offset++;
39 bufsize++;
40 }
41 if(cmp_offset_dst) {
42 *cmp_offset_dst = cmp_offset;
43 }
44 if (bufsize & 15) {
45 // In case we need to call invalidate (see below), we need an aligned buffer
46 bufsize += 16 - (bufsize & 15);
47 }
48 return bufsize;
49}
